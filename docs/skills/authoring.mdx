A contribution has two parts: an instruction file checked into `apps/skills/skills/` and a catalog record in `apps/skills/src/data/skills.ts`. Adding only the file makes it installable through the shell installer. Adding the record makes it browsable on [skills.n3wth.com](https://skills.n3wth.com).

<Steps>
  <Step title="Pick an ID and task">
    Choose a unique lowercase-hyphenated ID, such as `api-testing` or `react-patterns`. Focus on one well-defined task so the skill stays narrow and reliable.
  </Step>
  <Step title="Choose a layout">
    Decide between a flat file and a folder skill:

    - **Flat skill**: `skills/<id>.md` for instruction-only skills.
    - **Folder skill**: `skills/<id>/SKILL.md` for skills that need helper files or assets.
  </Step>
  <Step title="Write the instructions">
    Use the `SKILL.md` format. Start with YAML frontmatter, then add triggers, instructions, and validation prompts.

    ```markdown title="skills/my-skill/SKILL.md"
    ---
    name: my-skill
    description: What the skill does, and when to use it.
    ---
    
    # My Skill
    
    ## Triggers
    
    Use this skill when the user wants to:
    - Perform a specific task
    - Follow a defined workflow
    
    Keywords: "keyword1", "keyword2"
    
    ## Instructions
    
    1. Read the project manifest.
    2. Apply the relevant conventions.
    3. Include an example.
    4. Run checks and report failures.
    
    ## Validation
    
    Try: A matching prompt.
    Check: The output follows the instructions.
    ```
  </Step>
  <Step title="Add a catalog record">
    Open `apps/skills/src/data/skills.ts` and add a `Skill` object to the `skills` array. Here is a real example from the source:

    ```typescript
    {
      id: 'pdf',
      name: 'PDF Toolkit',
      description: 'Extract text and tables from PDFs, create documents, merge and split files, and process forms.',
      longDescription: 'Work with PDF documents programmatically. Extract text and tables, create new PDFs, merge documents, split pages, and process fillable forms.',
      category: 'documents',
      tags: ['pdf', 'documents', 'extraction'],
      icon: '▣',
      color: 'oklch(0.60 0.15 25)',
      skillFile: `${GITHUB_RAW_BASE}/pdf.md`,
      features: [
        'Text and table extraction',
        'PDF creation from scratch',
        'Document merging and splitting',
        'Form field processing',
        'Page manipulation and reordering'
      ],
      useCases: [
        'Invoice data extraction',
        'Report generation',
        'Document consolidation',
        'Form automation',
        'Contract processing'
      ],
      compatibility: ['gemini'],
      version: '1.0.0',
      lastUpdated: '2025-11-15'
    }
    ```

    For folder skills, set `skillFile` to `${GITHUB_RAW_BASE}/<id>/SKILL.md`.
  </Step>
  <Step title="Test locally">
    Run the shell installer against a staging directory and verify the file lands correctly:

    ```bash
    SKILLS_INSTALL_HOME=/tmp/staging bash /tmp/install.sh gemini my-skill
    ls /tmp/staging/.gemini/skills/
    ```
  </Step>
  <Step title="Open a pull request">
    Push your branch and open a PR against `n3wth/n3wth`. Include both the skill file and the catalog record so the skill appears on the website.
  </Step>
</Steps>

## Skill interface reference

Every catalog record must satisfy the `Skill` interface defined in `src/data/skills.ts`.

<ParamField path="id" type="string" required>
  Unique lowercase-hyphenated identifier.
</ParamField>

<ParamField path="name" type="string" required>
  Human-readable name shown in the catalog.
</ParamField>

<ParamField path="description" type="string" required>
  Short summary for cards and listings.
</ParamField>

<ParamField path="longDescription" type="string">
  Extended description for the detail page.
</ParamField>

<ParamField path="category" type="CategoryId" required>
  One of: `development`, `documents`, `creative`, `productivity`, `business`.
</ParamField>

<ParamField path="tags" type="string[]" required>
  Searchable keywords.
</ParamField>

<ParamField path="featured" type="boolean">
  Whether the skill appears in the featured section.
</ParamField>

<ParamField path="icon" type="string" required>
  A single-character or short icon string.
</ParamField>

<ParamField path="color" type="string" required>
  An OKLCH color value for theming.
</ParamField>

<ParamField path="features" type="string[]">
  List of capability bullets.
</ParamField>

<ParamField path="useCases" type="string[]">
  List of typical usage scenarios.
</ParamField>

<ParamField path="compatibility" type="AssistantId[]">
  Supported assistants. Currently only `['gemini']`.
</ParamField>

<ParamField path="version" type="string" required>
  Semantic version of the skill.
</ParamField>

<ParamField path="lastUpdated" type="string" required>
  ISO date of the last change.
</ParamField>

<ParamField path="contributor" type="Contributor">
  Optional attribution with `name`, `github`, and `url`.
</ParamField>

<ParamField path="samplePrompts" type="SamplePrompt[]">
  Example prompts and expected outputs for the detail page.
</ParamField>

<ParamField path="skillFile" type="string">
  Raw GitHub URL to the downloadable file or `SKILL.md`.
</ParamField>

<ParamField path="seoTitle" type="string">
  Custom page title for SEO.
</ParamField>

<ParamField path="seoDescription" type="string">
  Custom meta description for SEO.
</ParamField>

<ParamField path="relatedSkillIds" type="string[]">
  IDs of related skills for cross-linking.
</ParamField>

<ParamField path="faq" type="SkillFaq[]">
  Questions and answers for the detail page. Each entry can include `links`.
</ParamField>

<ParamField path="extraInstallCommands" type="ExtraInstallCommand[]">
  Additional install commands for other assistants. Used by `skill-creator` to provide a Claude Code recipe.
</ParamField>

<ParamField path="installHint" type="string">
  Extra guidance shown on the install page.
</ParamField>
